import { useState } from 'react';
import SuggestionPost from './SuggestionPost';

export default function SuggestionFeed() {
  const [suggestions, setSuggestions] = useState([
    {
      id: 1,
      title: 'Persistent slide-out cart menu',
      category: 'UI/UX Improvement',
      description: 'Keep the cart item list accessible without forcing page reloads.',
      createdAt: '2/10/2026',
    },
  ]);

  const handleAddSuggestion = (newSuggestion) => {
    setSuggestions((prev) => [newSuggestion, ...prev]);
  };

  return (
    <div className="mx-auto max-w-3xl p-6">
      <SuggestionPost onAddSuggestion={handleAddSuggestion} />

      <div className="space-y-4">
        <h3 className="text-md font-bold text-gray-800">Community Suggestions ({suggestions.length})</h3>
        {suggestions.length === 0 ? (
          <p className="text-sm text-gray-500">No suggestions submitted yet.</p>
        ) : (
          suggestions.map((item) => (
            <div key={item.id} className="rounded-xl border border-gray-200 bg-white p-5 shadow-sm">
              <div className="mb-2 flex items-center justify-between">
                <span className="rounded-full bg-gray-100 px-3 py-1 text-xs font-semibold text-gray-600">
                  {item.category}
                </span>
                <span className="text-xs text-gray-400">{item.createdAt}</span>
              </div>
              <h4 className="text-base font-bold text-gray-900">{item.title}</h4>
              <p className="mt-1 text-sm text-gray-600">{item.description}</p>
            </div>
          ))
        )}
      </div>
    </div>
  );
}
